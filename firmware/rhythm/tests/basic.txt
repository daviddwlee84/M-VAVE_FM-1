# RHYTHM in the host simulator: just fw-test rhythm (screenshots and audio in firmware/rhythm/build/host/)
# The host has no audio queue and no debounce, so a tap lands where the script puts it (LAT 0).
# hint numbers (rhythm.h): 1 GOOD, 3 MISSED NOTES, 5 SYNCOPATION SQUARED, 6 LATE AFTER RESTS, 8 RUSHING, 9 DRAGGING
wait 600
shot boot.png
expect page 1
expect bpm 90
expect bpb 4
expect echo 0
expect running 0

# ---- ECHO: a fixed rhythm at 120 BPM, tapped exactly
turn SELECT 30
expect bpm 120
pattern q_ee_r_q
expect notes 4
expect_text pattern q_ee_r_q
autotap 0
tap REC
expect running 1
expect echo 1
wait 2100
expect echo 2
shot echo_listen.png
peakreset
wav echo.wav
wait 2000
expect echo 3
expect peak_db_min -30
wait 2000
expect echo 4
shot echo_tap.png
wait 2400
wavstop
expect echo 5
expect running 0
expect taps 4
expect good 4
expect missed 0
expect score 100
expect hint 1
expect_text hint GOOD
shot echo_good.png

# ---- the same rhythm, every tap 45 ms early: RUSHING
autotap -45
tap REC
wait 8600
expect echo 5
expect taps 4
expect mean -45
expect_text hint RUSHING
shot echo_rushing.png

# ---- 45 ms late: DRAGGING
autotap 45
tap REC
wait 8600
expect mean 45
expect_text hint DRAGGING

# ---- only the note after the rest late: LATE AFTER RESTS
autotap 0 -1 3 60
tap REC
wait 8600
expect_text hint LATE_AFTER_RESTS
shot echo_late_rest.png

# ---- a note missed (the second): the others still line up
autotap 0 1
tap REC
wait 8600
expect missed 1
expect_text hint MISSED_NOTES
shot echo_missed.png

# ---- syncopation pulled onto the beat (at 90 BPM the CO is 333 ms off the beat)
turn SELECT -30
expect bpm 90
pattern q_syn_q
autotap 0 -1 2 150
tap REC
wait 11300
expect echo 5
expect_text hint SYNCOPATION_SQUARED
shot echo_synco.png

# ---- a generated rhythm, two bars, level 4 (up to TIKA-TIKA), tapped on time
turn K2 3
expect level 4
turn K3 1
expect bars 2
tap PLAY
expect running 1
expect echo 1
tap PLAY
expect running 0
expect echo 0
tap PLAY
wait 200
# (autotap needs the rhythm first: stop, plan it, then REC plays the same one)
tap PLAY
autotap 0
tap REC
wait 16600
expect echo 5
expect missed 0
expect score 100
shot echo_level4.png
